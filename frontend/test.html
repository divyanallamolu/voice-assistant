// ------------------------------- // VOICE RECOGNITION // ------------------------------- const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; let isListening = false; if (!SpeechRecognition) { alert("Speech recognition is not supported in this browser."); } else { const recognition = new SpeechRecognition(); recognition.lang = "en-US"; recognition.continuous = false; recognition.interimResults = false; recognition.onstart = () => { isListening = true; console.log("Listening..."); voiceButton.textContent = "🛑 Listening..."; status.textContent = "🎤 Listening..."; }; recognition.onresult = (event) => { const transcript = event.results[0][0].transcript; console.log("You said:", transcript); messageInput.value = transcript; // Send voice input to backend if ( socket && socket.readyState === WebSocket.OPEN ) { socket.send(transcript); } else { alert("Connect to the backend first!"); } }; recognition.onend = () => { isListening = false; console.log("Stopped listening"); voiceButton.textContent = "🎤 Start Listening"; // Don't change status if WebSocket is still connected if ( socket && socket.readyState === WebSocket.OPEN ) { status.textContent = "🟢 Connected"; } }; recognition.onerror = (event) => { isListening = false; console.error( "Speech recognition error:", event.error ); voiceButton.textContent = "🎤 Start Listening"; status.textContent = "❌ Voice Error: " + event.error; }; voiceButton.addEventListener("click", () => { if (isListening) { recognition.stop(); } else { recognition.start(); } }); }
